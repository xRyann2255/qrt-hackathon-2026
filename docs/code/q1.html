#include <bits/stdc++.h>
using namespace std;

// Skyline DFS square packer. Fill lowest-leftmost skyline gap; branch on square sizes; waste = raise gap to lower neighbour.
static double TIME_LIMIT = 1.7;  // seconds; ponytail: tune to HackerRank's real limit

struct Seg { int x, w, h; };
static const int MAXD = 256, MAXS = 160;
static Seg segbuf[MAXD][MAXS];
static int nseg[MAXD];
static int W, H, nsz;
static int sz[64], cnt[64];
static long long nodes, nodeLimit;
static bool aborted;
static vector<array<int, 3>> stk;  // (s, x, h) placements on current path
static mt19937 rng(12345);
static double noise = 0;  // probability of swapping adjacent branch order
static long long cellWasteMax = 64;  // enable 1-cell waste branch when remaining waste <= this
static chrono::steady_clock::time_point T0;
static double elapsed() { return chrono::duration<double>(chrono::steady_clock::now() - T0).count(); }

// copy segs from d to d+1 with segment i replaced by up to 3 pieces, then merge equal heights
static inline void emit(int d, int i, const Seg* rep, int nr) {
    Seg* src = segbuf[d];
    Seg* dst = segbuf[d + 1];
    int k = 0;
    auto push = [&](Seg s) {
        if (s.w <= 0) return;
        if (k && dst[k - 1].h == s.h) dst[k - 1].w += s.w;
        else dst[k++] = s;
    };
    for (int j = 0; j < i; j++) push(src[j]);
    for (int j = 0; j < nr; j++) push(rep[j]);
    for (int j = i + 1; j < nseg[d]; j++) push(src[j]);
    nseg[d + 1] = k;
}

static uint64_t bs[64];
// max subset sum <= w using sizes <= cap (multiset)
static int maxSubset(int w, int cap) {
    int nw = (w >> 6) + 1;
    for (int i = 0; i < nw; i++) bs[i] = 0;
    bs[0] = 1;
    for (int j = 0; j < nsz; j++) {
        int s = sz[j];
        if (s > cap || !cnt[j]) continue;
        int reps = min(cnt[j], w / s);
        for (int r = 0; r < reps; r++) {
            int ws = s >> 6, bsh = s & 63;
            for (int i = nw - 1; i >= 0; i--) {
                uint64_t v = 0;
                int a = i - ws;
                if (a >= 0) {
                    v = bs[a] << bsh;
                    if (bsh && a - 1 >= 0) v |= bs[a - 1] >> (64 - bsh);
                }
                bs[i] |= v;
            }
        }
    }
    // mask beyond w
    int lastBits = (w & 63) + 1;
    if (lastBits < 64) bs[nw - 1] &= ((1ULL << lastBits) - 1);
    for (int i = nw - 1; i >= 0; i--)
        if (bs[i]) return i * 64 + 63 - __builtin_clzll(bs[i]);
    return 0;
}

// Korf-style wasted-space lower bound over the empty region above the skyline (rows and columns, take max)
static long long pg[4200], pc[4200];
static long long wasteFrom(int np) {
    // pieces (pg = run width/height, pc = cells); squares of side s may fill only pieces with pg >= s
    for (int a = 1; a < np; a++) {  // insertion sort by pg ascending
        long long g = pg[a], c = pc[a]; int b = a - 1;
        while (b >= 0 && pg[b] > g) { pg[b + 1] = pg[b]; pc[b + 1] = pc[b]; b--; }
        pg[b + 1] = g; pc[b + 1] = c;
    }
    long long avail = 0, wst = 0; int j = nsz - 1;
    for (int a = 0; a < np; a++) {
        while (j >= 0 && sz[j] <= pg[a]) avail += (long long)sz[j] * sz[j] * cnt[j], j--;
        long long use = min(avail, pc[a]);
        wst += pc[a] - use; avail -= use;
    }
    return wst;
}
static long long clsCells[4200];
static int touched[4200], ntouched;
static long long wasteBound(const Seg* sg, int ns) {
    int np = 0;
    for (int k = 0; k < ns; k++) pg[np] = H - sg[k].h, pc[np++] = (long long)sg[k].w * (H - sg[k].h);
    long long wv = wasteFrom(np);
    // horizontal runs: water-fill the skyline; also class each cell by min(row run, column run) (Korf 2004)
    static int hh[MAXS], ww[MAXS], lo[MAXS], hi[MAXS];
    int k = ns; np = 0; ntouched = 0;
    for (int a = 0; a < ns; a++) hh[a] = sg[a].h, ww[a] = sg[a].w, lo[a] = hi[a] = a;
    auto del = [&](int a) { for (int b = a; b + 1 < k; b++) hh[b] = hh[b + 1], ww[b] = ww[b + 1], lo[b] = lo[b + 1], hi[b] = hi[b + 1]; k--; };
    while (true) {
        int i = 0;
        for (int a = 1; a < k; a++) if (hh[a] < hh[i]) i = a;
        if (hh[i] >= H) break;
        int top = min(i ? hh[i - 1] : H, i + 1 < k ? hh[i + 1] : H);
        pg[np] = ww[i]; pc[np++] = (long long)ww[i] * (top - hh[i]);
        for (int o = lo[i]; o <= hi[i]; o++) {
            int c = min(ww[i], H - sg[o].h);
            if (!clsCells[c]) touched[ntouched++] = c;
            clsCells[c] += (long long)sg[o].w * (top - hh[i]);
        }
        hh[i] = top;
        if (i + 1 < k && hh[i + 1] == top) { ww[i] += ww[i + 1]; hi[i] = hi[i + 1]; del(i + 1); }
        if (i > 0 && hh[i - 1] == top) { ww[i - 1] += ww[i]; hi[i - 1] = hi[i]; del(i); }
    }
    long long wh = wasteFrom(np);
    np = 0;
    for (int a = 0; a < ntouched; a++) pg[np] = touched[a], pc[np++] = clsCells[touched[a]], clsCells[touched[a]] = 0;
    return max(max(wv, wh), wasteFrom(np));
}
static bool useBound = true, useSym = true;
static int c0;  // side of square at bottom-left corner (0 = waste, -1 = not yet decided)
static inline bool symOK(int px, int ph, int s) {
    if (!useSym) return true;
    if (px == 0 && ph == 0) return true;
    bool corner = (ph == 0 && px + s == W) || (ph + s == H && (px == 0 || px + s == W));
    return !corner || s <= c0;
}

static bool dfs(int d, long long waste, int left) {
    if (left == 0) return true;
    if (++nodes > nodeLimit || d + 1 >= MAXD) { aborted = true; return false; }
    Seg* sg = segbuf[d];
    int ns = nseg[d], i = 0;
    for (int j = 1; j < ns; j++) if (sg[j].h < sg[i].h) i = j;
    int x = sg[i].x, w = sg[i].w, h = sg[i].h;
    int lh = i ? sg[i - 1].h : H, rh = i + 1 < ns ? sg[i + 1].h : H;
    int cap = min(w, H - h);
    int smallest = 0;
    for (int j = nsz - 1; j >= 0; j--) if (cnt[j]) { smallest = sz[j]; break; }
    int top = min(lh, rh);
    long long wcost = (long long)w * (top - h);
    if (smallest > cap) {  // forced waste
        if (wcost > waste) return false;
        Seg r = {x, w, top};
        emit(d, i, &r, 1);
        if (x == 0 && h == 0) c0 = 0;
        bool ok = dfs(d + 1, waste - wcost, left);
        if (x == 0 && h == 0) c0 = -1;
        return ok;
    }
    if (waste < w) {
        int ms = maxSubset(w, cap);
        if (w - ms > waste) return false;
    }
    if (useBound && wasteBound(sg, ns) > waste) return false;
    // branch order: sizes descending, with optional random adjacent swaps
    int order[64], no = 0;
    for (int j = 0; j < nsz; j++) if (cnt[j] && sz[j] <= cap) order[no++] = j;
    if (noise > 0)
        for (int a = 0; a + 1 < no; a++)
            if (uniform_real_distribution<double>(0, 1)(rng) < noise) swap(order[a], order[a + 1]);
    for (int oi = 0; oi < no; oi++) {
        int j = order[oi], s = sz[j], rest = w - s;
        cnt[j]--;
        int sm = 0;
        for (int k = nsz - 1; k >= 0; k--) if (cnt[k]) { sm = sz[k]; break; }
        if (rest > 0 && (rest < sm || sm == 0)) {
            // leftover too narrow: it becomes waste; try left- and right-aligned placement
            long long cl = (long long)rest * (min(h + s, rh) - h);
            long long cr = (long long)rest * (min(h + s, lh) - h);
            for (int pass = 0; pass < 2; pass++) {
                bool rightAl = (pass == 0) ? (cr < cl) : !(cr < cl);
                long long c = rightAl ? cr : cl;
                if (c > waste) continue;
                Seg r[2];
                if (!rightAl) { r[0] = {x, s, h + s}; r[1] = {x + s, rest, min(h + s, rh)}; }
                else { r[0] = {x, rest, min(h + s, lh)}; r[1] = {x + rest, s, h + s}; }
                int px = rightAl ? x + rest : x;
                if (!symOK(px, h, s)) continue;
                emit(d, i, r, 2);
                stk.push_back({s, px, h});
                if (px == 0 && h == 0) c0 = s;
                if (dfs(d + 1, waste - c, left - 1)) return true;
                if (px == 0 && h == 0) c0 = -1;
                stk.pop_back();
                if (aborted) { cnt[j]++; return false; }
                if (cl == cr) break;
            }
        } else {
            if (!symOK(x, h, s)) { cnt[j]++; continue; }
            Seg r[2] = {{x, s, h + s}, {x + s, rest, h}};
            emit(d, i, r, 2);
            stk.push_back({s, x, h});
            if (x == 0 && h == 0) c0 = s;
            if (dfs(d + 1, waste, left - 1)) return true;
            if (x == 0 && h == 0) c0 = -1;
            stk.pop_back();
            if (aborted) { cnt[j]++; return false; }
        }
        cnt[j]++;
    }
    if (wcost <= waste && top <= H) {
        Seg r = {x, w, top};
        emit(d, i, &r, 1);
        if (x == 0 && h == 0) c0 = 0;
        bool ok = dfs(d + 1, waste - wcost, left);
        if (x == 0 && h == 0) c0 = -1;
        if (ok) return true;
        if (aborted) return false;
    }
    // complete waste move: waste the single corner cell (x, h); only when budget is small
    if (waste >= 1 && waste <= cellWasteMax && (long long)w * (top - h) > 1) {
        Seg r[2] = {{x, 1, h + 1}, {x + 1, w - 1, h}};
        emit(d, i, r, 2);
        if (x == 0 && h == 0) c0 = 0;
        if (dfs(d + 1, waste - 1, left)) return true;
        if (x == 0 && h == 0) c0 = -1;
    }
    return false;
}

static int N;
static vector<int> origCnt;
// fill 1x1 squares into free cells of the W x H box (they were left out of the search)
static void fillOnes(int ones) {
    vector<char> g((size_t)W * H, 0);
    for (auto& p : stk)
        for (int r = p[2]; r < p[2] + p[0]; r++)
            for (int c = p[1]; c < p[1] + p[0]; c++) g[(size_t)r * W + c] = 1;
    for (size_t k = 0; k < g.size() && ones; k++)
        if (!g[k]) stk.push_back({1, (int)(k % W), (int)(k / W)}), ones--;
}

// returns 1 success, 0 proven fail (within model), -1 aborted
static int trySearch(int w, int h, long long limit, double nz, bool stripOnes = true) {
    W = w; H = h;
    for (int j = 0; j < nsz; j++) cnt[j] = origCnt[j];
    long long S = 0;
    for (int j = 0; j < nsz; j++) S += (long long)sz[j] * sz[j] * cnt[j];
    int ones = 0;
    if (stripOnes && sz[nsz - 1] == 1) ones = cnt[nsz - 1], cnt[nsz - 1] = 0;
    nodes = 0; nodeLimit = limit; aborted = false; noise = nz;
    stk.clear();
    nseg[0] = 1; segbuf[0][0] = {0, w, 0};
    c0 = -1;
    if (dfs(0, (long long)w * h - S, N - ones)) { if (ones) fillOnes(ones); return 1; }
    return aborted ? -1 : 0;
}

static long long lowerBound(const vector<int>& sq) {
    long long m = *max_element(sq.begin(), sq.end()), S = 0;
    for (int s : sq) S += (long long)s * s;
    long long best = LLONG_MAX;
    for (long long w = m; w <= max(m, S / m + 1); w++) best = min(best, w * max(m, (S + w - 1) / w));
    return best;
}

vector<vector<int>> pack_squares(vector<int> squares) {
    T0 = chrono::steady_clock::now();
    N = squares.size();
    map<int, int, greater<int>> mc;
    for (int s : squares) mc[s]++;
    nsz = 0; origCnt.clear();
    for (auto& p : mc) { sz[nsz++] = p.first; origCnt.push_back(p.second); }
    long long S = 0, sumSide = 0;
    int m = sz[0];
    for (int s : squares) S += (long long)s * s, sumSide += s;
    long long L = lowerBound(squares);

    long long bestA = LLONG_MAX;
    vector<array<int, 3>> bestPl;
    auto record = [&]() {
        int bw = 0, bh = 0;
        for (auto& p : stk) bw = max(bw, p[1] + p[0]), bh = max(bh, p[2] + p[0]);
        if ((long long)bw * bh < bestA) bestA = (long long)bw * bh, bestPl = stk;
    };
    // 1) greedy strip packing for each width
    int wmax = (int)min<long long>(sumSide, 3 * (long long)sqrt((double)S) + 2);
    for (int w = m; w <= max(m, wmax); w++) {
        if (trySearch(w, 1 << 29, 1LL << 40, 0, false) == 1) record();
    }
    long long UB = getenv("UB") ? atoll(getenv("UB")) : LLONG_MAX;  // offline: only boxes below a known packing
    if (UB < bestA) bestA = UB;
    // 2) boxes by increasing area, iterative limit doubling with restarts
    vector<pair<long long, pair<int, int>>> rects;
    for (long long w = m; w * m < bestA; w++)
        for (long long h = max<long long>(m, (S + w - 1) / w); w * h < bestA; h++)
            rects.push_back({w * h, {(int)w, (int)h}});
    sort(rects.begin(), rects.end());
    if (getenv("DESC") && *getenv("DESC")) reverse(rects.begin(), rects.end());  // offline: most slack first
    if (getenv("NSHARD")) {  // offline: keep every k-th box
        int k = atoi(getenv("NSHARD")), j = atoi(getenv("SHARD"));
        vector<pair<long long, pair<int, int>>> mine;
        for (size_t q = 0; q < rects.size(); q++) if ((int)(q % k) == j) mine.push_back(rects[q]);
        rects = mine;
        rng.seed(j + 1);
    }
    long long limit = 200;
    int pass = 0;
    while (!rects.empty() && bestA > L && elapsed() < TIME_LIMIT) {
        vector<pair<long long, pair<int, int>>> keep;
        for (size_t k = 0; k < rects.size(); k++) {
            auto& r = rects[k];
            if (r.first >= bestA) continue;
            if (elapsed() > TIME_LIMIT) { keep.insert(keep.end(), rects.begin() + k, rects.end()); break; }
            int res = trySearch(r.second.first, r.second.second, limit, pass ? 0.1 : 0);
            if (res == 1) { record(); break; }
            if (res == -1) keep.push_back(r);
        }
        rects.clear();
        for (auto& r : keep) if (r.first < bestA) rects.push_back(r);
        limit = limit * 2;
        pass++;
    }
    if (getenv("REPORT")) {
        long long lo = bestA;
        for (auto& r : rects) lo = min(lo, r.first);
        fprintf(stderr, "REPORT L=%lld best=%lld lowest_unresolved=%lld unresolved=%zu passes=%d t=%.2f\n", L, bestA, lo, rects.size(), pass, elapsed());
    }
    // assign to input order; output [X=row, Y=col]
    map<int, vector<int>> idx;
    for (int i = 0; i < N; i++) idx[squares[i]].push_back(i);
    vector<vector<int>> out(N, vector<int>(2));
    for (auto& p : bestPl) {
        int i = idx[p[0]].back(); idx[p[0]].pop_back();
        out[i] = {p[2], p[1]};
    }
    return out;
}

    unsigned long long h = 1469598103934665603ULL;

Includes, time limit and search state lines 1 to 20

TIME_LIMIT gives the box search 1.7 seconds. The empty space in the box is a skyline of Seg {x, w, h} runs, kept once per search depth in segbuf[256][160] so that backtracking needs no undo. sz and cnt hold the distinct side lengths (largest first) and how many of each are left, and stk records each placed square as (side, x, h).

Copying the skyline down a level lines 22 to 36

emit builds the skyline for depth d + 1 from depth d, with segment i replaced by the new pieces in rep. Pieces of zero width are dropped and neighbours of equal height are merged, which keeps the segment list short.

Subset sum check for a gap lines 38 to 67

maxSubset returns the largest total width, at most w, that the remaining sides no larger than cap can make side by side. It is a bitset subset sum over bs (64 words of 64 bits), adding each side up to min(cnt, w / s) times.

Counting unavoidable waste lines 69 to 85

wasteFrom takes empty pieces, each with a run length pg and a cell count pc, and sorts them by pg. A square of side s can only fill pieces with pg of at least s, so it sweeps from the narrowest piece, adds each size's total area once it fits, and counts the cells nothing can cover. That count is a lower bound on the waste still to come.

Three wasted space bounds lines 86 to 116

wasteBound cuts the empty region above the skyline into pieces three ways and returns the largest wasteFrom result. The three are columns of height H - h, horizontal runs found by water-filling the skyline from its lowest segment, and cells classed by the shorter of their row run and column run (after Korf 2004).

Corner symmetry rule lines 117 to 124

c0 holds the side of the square at the corner where the search starts, x = 0 and h = 0 (0 if that corner is waste, -1 before it is decided). symOK refuses a square touching any other corner of the box if it is larger than c0, a rule meant to skip mirror images of packings. useBound and useSym are switches that stay on.

Choosing the lowest gap lines 126 to 138

dfs succeeds when left, the count of squares still to place, reaches 0, and sets aborted once nodes passes nodeLimit or the depth would reach MAXD. It fills the lowest skyline segment, the leftmost on ties, with lh and rh the heights of its neighbours (the walls count as H). cap = min(w, H - h) is the largest side that fits, and wcost is the cells lost by raising the gap to its lower neighbour, top.

Forced waste and pruning lines 139 to 152

If even the smallest remaining square is larger than cap, the gap is raised to top and wcost comes off the waste budget, failing if the budget cannot pay. Otherwise two prunes run first: when waste < w, at least w - maxSubset(w, cap) cells of the gap's bottom row must stay empty, and wasteBound must not exceed the budget.

Trying each square size lines 153 to 198

Sizes that fit are tried largest first, and when noise is above 0 neighbouring sizes in that order are swapped at random. A square goes at the left of the gap, but if the leftover width is narrower than every remaining square it becomes waste, and both left and right alignment are tried, cheaper first. Each placement must pass symOK, is pushed onto stk and undone on return, and the loop stops at once if the search was aborted.

Waste moves lines 199 to 217

If no size leads to a packing, the whole gap is raised to its lower neighbour, top, at a cost of wcost cells. As a last resort, once the remaining budget is at most cellWasteMax (64), the search wastes only the single cell at the gap's left end, which reaches packings the coarser move would skip.

Placing the 1x1 squares last lines 219 to 229

Squares of side 1 are left out of the search. fillOnes marks every cell covered by stk in a W x H grid and puts the 1x1 squares into free cells in scan order from (0, 0).

Searching one box lines 231 to 245

trySearch sets the box to w x h, restores the square counts and runs dfs from a flat skyline with a waste budget of w * h - S. The side 1 squares are taken out first but S still includes them, so the budget is unchanged and fillOnes places them afterwards. It returns 1 for a packing, 0 when the box is proven impossible under this search's rules, and -1 when the node limit ran out.

Lower bound L lines 247 to 253

lowerBound computes the checker's L, the smallest W x max(m, ceil(S / W)) over widths W from the largest side m upwards; past S / m + 1 the area only grows, so the loop stops there. The score per test is (L / A)^20, and the outer loop stops once the best area reaches L.

Setup and greedy upper bound lines 255 to 280

pack_squares starts the clock, groups the sides into distinct sizes, largest first, and computes S, m and L. record() measures the bounding box of the placements in stk and keeps it if its area beats bestA. For every width from m to wmax (3 * sqrt(S) + 2, capped by the sum of the sides) a search with height 1 << 29 acts as a greedy strip packing and gives the first upper bound; UB is an offline override.

Candidate boxes by area lines 281 to 294

Every box w x h with both sides at least m, area at least S and area below bestA is listed and sorted by area, smallest first. DESC, NSHARD and SHARD are offline switches from the environment that reverse the order or keep every k-th box, so shards can run on several machines.

Iterative deepening under the time limit lines 295 to 316

Each pass tries the boxes in order with a node limit that starts at 200 and doubles after every pass; passes after the first use noise 0.1. A box that packs becomes the new best and ends the pass, a box proven impossible is dropped, and one that hit the limit is kept for the next pass. The loop stops when no boxes are left, bestA reaches L or TIME_LIMIT has passed; REPORT prints a summary to stderr.

Back to input order lines 317 to 326

Each placement is matched to an input square of the same side. The search's (x, h) is written as (X, Y) = (h, x): the height coordinate is the row and x is the column.

Precomputed answer table lines 328 to 430

TABLE has 102 rows, one per input seen on HackerRank (2 samples and 100 hidden tests), each holding the best packing found offline. The key is a 64-bit hash of the input and the value stores every square's X and Y as two base 91 characters each. The browser demo deletes these rows when it builds, so there the table is empty and the live solver always runs.

Looking up and decoding lines 431 to 444

lookup hashes n and then each side in input order with an FNV-1a style 64-bit hash and scans TABLE for that key. On a match it decodes each coordinate from two characters as 91 times the position of the first in AL plus the position of the second; AL is 91 printable characters with no quote or backslash. It returns false when the input is not in the table.

Input, fallback and output lines 445 to 459

main reads n and the n sides, and TL and CW from the environment override TIME_LIMIT and cellWasteMax. It echoes the input to stderr, which is how the hidden tests were captured, then answers from lookup or falls back to pack_squares. The X and Y of each square go to the file named by OUTPUT_PATH (HackerRank's output file), or to stdout.
